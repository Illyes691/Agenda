    // Génère l'URL officielle Google Calendar
    function openGoogleCalendar(ev) {
      let baseUrl = "https://calendar.google.com/calendar/render?action=TEMPLATE";
      baseUrl += "&text=" + encodeURIComponent(ev.title);
      baseUrl += "&location=" + encodeURIComponent(ev.location);

      // Si l'utilisateur a saisi une heure dans le bandeau
      const startInput = document.getElementById('quickStart').value;
      if (startInput) {
        const start = new Date(startInput);
        const end = new Date(start.getTime() + (ev.duration || 60) * 60000);
        
        const formatUTC = (d) => d.toISOString().replace(/[-:]/g, '').split('.')[0] + 'Z';
        baseUrl += "&dates=" + formatUTC(start) + "/" + formatUTC(end);
      }

      // Ouvre Google Calendar dans un nouvel onglet avec tout pré-rempli
      window.open(baseUrl, '_blank');
    }

    // Affichage des colonnes et des cartes
    function render() {
      const container = document.getElementById('board');
      container.innerHTML = '';

      categories.forEach(cat => {
        const col = document.createElement('div');
        col.className = 'col';

        col.innerHTML = `
          <div class="col-header">
            <span class="col-dot" style="background-color: ${cat.color}"></span>
            <span>${cat.name}</span>
          </div>
        `;

        cat.events.forEach(ev => {
          const card = document.createElement('button');
          card.type = 'button';
          card.className = 'card';
          card.style.setProperty('--col-color', cat.color);

            <div class="card-meta">
              <span class="card-pill">📍 ${ev.location}</span>
              <span class="card-pill">⏱️ ${ev.duration} min</span>
            </div>
          `;

          card.onclick = () => openGoogleCalendar(ev);
          col.appendChild(card);
        });

        container.appendChild(col);
      });
    }
